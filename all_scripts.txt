df_events = pd.read_csv('athlete_events.csv', low_memory=False)
df_deporte = pd.read_csv('2_DEPORTE.csv')

def obtener_disciplina(fila):
    sport = str(fila['Sport'])
    event = str(fila['Event'])
    # 1. Quitar el nombre del deporte del inicio
    sin_deporte = re.sub(r'^' + re.escape(sport) + r'\s+', '', event)
    # 2. Quitar el genero
    disciplina_pura = re.sub(r"^(Men's\s+|Women's\s+|Mixed\s+)", "", sin_deporte)
    return disciplina_pura.strip()

df_base = df_events[['Sport', 'Event']].dropna().drop_duplicates()
df_base['NOMBRE'] = df_base.apply(obtener_disciplina, axis=1)

# Extraer disciplinas unicas asociadas a su deporte
df_disciplina = df_base[['NOMBRE', 'Sport']].drop_duplicates().reset_index(drop=True)

df_disciplina = df_disciplina.merge(df_deporte, left_on='Sport', right_on='NOMBRE', how='left', suffixes=('', '_dep'))
df_disciplina = df_disciplina.rename(columns={'ID': 'ID_DEPORTE'})
df_disciplina = df_disciplina[['NOMBRE', 'ID_DEPORTE']]

df_disciplina.insert(0, 'ID', range(1, len(df_disciplina) + 1))
df_disciplina.to_csv('6_DISCIPLINA.csv', index=False)

df_res['SEASON'] = df_res['Games'].str.extract(r'(Summer|Winter)').fillna('Unknown')

# --- INYECTOR AUTOMÁTICO DE OLIMPIADAS FALTANTES ---
olimpiadas_cat = df_olimpiada[['SEASON', 'YEAR']].drop_duplicates()
olimpiadas_res = df_res[['SEASON', 'YEAR']].drop_duplicates()

faltantes_olim = olimpiadas_res.merge(olimpiadas_cat, on=['SEASON', 'YEAR'], how='left', indicator=True)
faltantes_olim = faltantes_olim[faltantes_olim['_merge'] == 'left_only'][['SEASON', 'YEAR']]

if not faltantes_olim.empty:
    print(f"Detectadas {len(faltantes_olim)} Olimpiadas nuevas. Inyectándolas al catálogo 3_OLIMPIADA...")
    max_id_olim = df_olimpiada['ID'].max() if not df_olimpiada.empty else 0
    faltantes_olim['ID'] = range(max_id_olim + 1, max_id_olim + 1 + len(faltantes_olim))
    faltantes_olim = faltantes_olim[['ID', 'SEASON', 'YEAR']] # Ordenar columnas
    df_olimpiada = pd.concat([df_olimpiada, faltantes_olim], ignore_index=True)
    df_olimpiada.to_csv('3_OLIMPIADA.csv', index=False)
    print("Catálogo 3_OLIMPIADA.csv expandido exitosamente.")
# ---------------------------------------------------

# --- INYECTOR AUTOMÁTICO DE EVENTOS FALTANTES ---
eventos_cat = set(df_evento['NOMBRE'].unique())
eventos_res = df_res[['Event', 'Discipline']].dropna(subset=['Event']).drop_duplicates()

if not faltantes_evt.empty:

def limpiar_lugar(pos):
    if pd.isna(pos): return np.nan
    numeros = re.findall(r'\d+', str(pos))
    return int(numeros[0]) if numeros else np.nan

def obtener_estado(pos):
    if pd.isna(pos): return 'COMPLETED'
    pos_str = str(pos).upper()
    if 'DNS' in pos_str: return 'DNS' 
    if 'DNF' in pos_str: return 'DNF' 
    if 'DQ' in pos_str: return 'DQ'   
    return 'COMPLETED'

df_res['PLACE'] = df_res['Pos'].apply(limpiar_lugar)
df_res['STATE'] = df_res['Pos'].apply(obtener_estado)
df_res['TIED'] = df_res['Pos'].astype(str).str.contains('=')

# Cruzar con los catálogos que AHORA están 100% completos

df_res = df_res.merge(df_evento[['ID', 'NOMBRE']], left_on='Event', right_on='NOMBRE', how='left').rename(columns={'ID': 'ID_EVENTO'})
df_res['Medal'] = df_res['Medal'].fillna('No medal')
df_res = df_res.merge(df_medalla[['ID', 'NOMBRE']], left_on='Medal', right_on='NOMBRE', how='left').rename(columns={'ID': 'ID_MEDALLA'})

df_resultado = df_res[['TIED', 'PLACE', 'STATE', 'ID_MEDALLA', 'ID_OLIMPIADA', 'ID_EVENTO']].drop_duplicates().reset_index(drop=True)

df_resultado['ID_MEDALLA'] = df_resultado['ID_MEDALLA'].astype('Int64')
df_resultado['ID_OLIMPIADA'] = df_resultado['ID_OLIMPIADA'].astype('Int64')
df_resultado['ID_EVENTO'] = df_resultado['ID_EVENTO'].astype('Int64')

df_resultado.insert(0, 'ID', range(1, len(df_resultado) + 1))
df_resultado.to_csv('12_RESULTADO.csv', index=False)

print("12_RESULTADO.csv generado con el 100% de la información.")
